// ===================================================== // YAR AFGHANISTAN // JavaScript کامل اتصال ۵ بخش به Cloudflare API // // بخش‌ها: // 1. چت // 2. مترجم // 3. کمک درسی // 4. نوشتن // 5. ساخت آگهی // // API: // POST /api/chat // ===================================================== "use strict"; const CHAT_API_URL = "/api/chat"; // ===================================================== // 1. پاک‌سازی Thinking / Reasoning // ===================================================== function cleanAIResponse(text) { if (!text || typeof text !== "string") { return ""; } let result = text; // حذف ... result = result.replace( //gi, "" ); // حذف ... result = result.replace( //gi, "" ); // حذف ... result = result.replace( //gi, "" ); // حذف ... result = result.replace( //gi, "" ); // حذف تگ‌های احتمالی result = result.replace( /<\/?(think|thinking|reasoning|analysis)>/gi, "" ); // حذف Thinking: / Reasoning: / Analysis: result = result.replace( /^\s*(thinking|reasoning|analysis)\s*:\s*/i, "" ); // حذف Answer: / Final: result = result.replace( /^\s*(answer|final|پاسخ نهایی|پاسخ)\s*:\s*/i, "" ); // حذف Markdown code fence در ابتدا result = result.replace( /^```(?:text|markdown)?\s*/i, "" ); // حذف Markdown code fence در انتها result = result.replace( /\s*```$/i, "" ); return result.trim(); } // ===================================================== // 2. پیدا کردن container چت // ===================================================== function getChatContainer() { return ( document.getElementById("chatMessages") || document.getElementById("messages") || document.querySelector(".chat-messages") || document.querySelector("#chat") ); } // ===================================================== // 3. نمایش پیام // ===================================================== function addChatMessage(text, sender = "ai") { const container = getChatContainer(); if (!container) { console.warn("Chat container not found."); return null; } const message = document.createElement("div"); message.className = sender === "user" ? "message user-message" : "message ai-message"; const content = document.createElement("div"); content.className = "message-content"; // امنیت: // پاسخ AI مستقیماً به عنوان HTML اجرا نمی‌شود. content.textContent = text; message.appendChild(content); container.appendChild(message); container.scrollTop = container.scrollHeight; return message; } // ===================================================== // 4. Loading // ===================================================== function addLoadingMessage(text = "در حال پردازش...") { const container = getChatContainer(); if (!container) { return null; } const loading = document.createElement("div"); loading.className = "message ai-message loading-message"; const content = document.createElement("div"); content.className = "message-content"; content.textContent = text; loading.appendChild(content); container.appendChild(loading); container.scrollTop = container.scrollHeight; return loading; } // ===================================================== // 5. ارسال درخواست عمومی به API // ===================================================== async function callYarAPI(prompt, options = {}) { const { history = [], loadingText = "در حال پردازش..." } = options; const cleanPrompt = typeof prompt === "string" ? prompt.trim() : ""; if (!cleanPrompt) { throw new Error("پیام خالی است."); } const loading = addLoadingMessage(loadingText); try { const safeHistory = Array.isArray(history) ? history .filter( item => item && typeof item === "object" && ( item.role === "user" || item.role === "assistant" ) && typeof item.content === "string" ) .slice(-10) : []; const response = await fetch( CHAT_API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: cleanPrompt, messages: safeHistory }) } ); let data; try { data = await response.json(); } catch { throw new Error( "پاسخ API معتبر نیست." ); } if (loading) { loading.remove(); } if ( !response.ok || data?.success === false ) { const errorMessage = data?.error || "ارتباط با هوش مصنوعی برقرار نشد."; throw new Error(errorMessage); } let answer = data?.reply; if ( typeof answer !== "string" || !answer.trim() ) { answer = data?.message; } if ( typeof answer !== "string" || !answer.trim() ) { throw new Error( "هوش مصنوعی پاسخ قابل استفاده‌ای برنگرداند." ); } answer = cleanAIResponse(answer); if (!answer) { throw new Error( "پاسخ نهایی خالی است." ); } return answer; } catch (error) { if (loading) { loading.remove(); } console.error( "Yar API Error:", error ); throw error; } } // ===================================================== // 6. نمایش خطا // ===================================================== function showAIError(error) { const message = error?.message || "ارتباط با هوش مصنوعی برقرار نشد."; addChatMessage( "❌ " + message, "ai" ); } // ===================================================== // 7. بخش چت // ===================================================== async function sendMessageToAI(message) { const userMessage = typeof message === "string" ? message.trim() : ""; if (!userMessage) { return; } addChatMessage( userMessage, "user" ); try { const answer = await callYarAPI( userMessage, { loadingText: "در حال پاسخ..." } ); addChatMessage( answer, "ai" ); } catch (error) { showAIError(error); } } // ===================================================== // 8. مترجم // ===================================================== async function translateText() { const input = document.getElementById( "translateInput" ); const source = document.getElementById( "sourceLanguage" ); const target = document.getElementById( "targetLanguage" ); const result = document.getElementById( "translateResult" ); if (!input) { console.warn( "translateInput not found." ); return; } const text = input.value.trim(); if (!text) { alert( "لطفاً متنی برای ترجمه وارد کنید." ); return; } const sourceLanguage = source?.value || "auto"; const targetLanguage = target?.value || "دری"; const prompt = ` Translate the following text. Source language: ${sourceLanguage} Target language: ${targetLanguage} Important: - Return ONLY the translation. - Do not explain. - Do not add thinking or reasoning. - Preserve the original meaning. Text: ${text} `; try { const answer = await callYarAPI( prompt, { loadingText: "در حال ترجمه..." } ); if (result) { result.textContent = answer; } // اگر نتیجه در چت هم وجود دارد if (!result) { addChatMessage( answer, "ai" ); } } catch (error) { if (result) { result.textContent = "❌ " + ( error?.message || "ترجمه انجام نشد." ); } else { showAIError(error); } } } // ===================================================== // 9. کمک درسی // ===================================================== async function solveStudyQuestion() { const input = document.getElementById( "studyInput" ); const subject = document.getElementById( "studySubject" ); const level = document.getElementById( "studyLevel" ); const result = document.getElementById( "studyResult" ); if (!input) { console.warn( "studyInput not found." ); return; } const question = input.value.trim(); if (!question) { alert( "لطفاً سوال درسی را وارد کنید." ); return; } const subjectName = subject?.value || "عمومی"; const studentLevel = level?.value || "نامشخص"; const prompt = ` You are an educational assistant. Subject: ${subjectName} Student level: ${studentLevel} Question: ${question} Answer in the same language as the question. Rules: - Explain clearly. - Use simple language. - Show steps when necessary. - Do not reveal thinking or internal reasoning. - Give only the useful final educational answer. `; try { const answer = await callYarAPI( prompt, { loadingText: "در حال حل سوال..." } ); if (result) { result.textContent = answer; } else { addChatMessage( answer, "ai" ); } } catch (error) { if (result) { result.textContent = "❌ " + ( error?.message || "پاسخ درسی دریافت نشد." ); } else { showAIError(error); } } } // ===================================================== // 10. بخش نوشتن // ===================================================== async function generateWriting() { const input = document.getElementById( "writingInput" ); const type = document.getElementById( "writingType" ); const result = document.getElementById( "writingResult" ); if (!input) { console.warn( "writingInput not found." ); return; } const idea = input.value.trim(); if (!idea) { alert( "لطفاً موضوع نوشته را وارد کنید." ); return; } const writingType = type?.value || "متن عمومی"; const prompt = ` You are a professional writing assistant. Writing type: ${writingType} User request: ${idea} Instructions: - Write a polished and useful final text. - Use the same language as the user. - Do not explain your process. - Do not show thinking or reasoning. - Return only the finished text. `; try { const answer = await callYarAPI( prompt, { loadingText: "در حال نوشتن..." } ); if (result) { result.textContent = answer; } else { addChatMessage( answer, "ai" ); } } catch (error) { if (result) { result.textContent = "❌ " + ( error?.message || "نوشتن انجام نشد." ); } else { showAIError(error); } } } // ===================================================== // 11. ساخت آگهی // ===================================================== async function generateAdvertisement() { const title = document.getElementById( "adTitle" ); const details = document.getElementById( "adDetails" ); const price = document.getElementById( "adPrice" ); const location = document.getElementById( "adLocation" ); const result = document.getElementById( "adResult" ); const adTitle = title?.value.trim() || ""; const adDetails = details?.value.trim() || ""; const adPrice = price?.value.trim() || ""; const adLocation = location?.value.trim() || ""; if (!adTitle && !adDetails) { alert( "لطفاً عنوان یا مشخصات آگهی را وارد کنید." ); return; } const prompt = ` Create a professional advertisement in the same language as the user. Title: ${adTitle} Details: ${adDetails} Price: ${adPrice || "تعیین نشده"} Location: ${adLocation || "تعیین نشده"} Requirements: - Create an attractive but honest advertisement. - Do not invent important facts. - Include a clear title. - Include useful details. - Include price and location if provided. - Make it ready to publish. - Do not show thinking or reasoning. - Return only the final advertisement. `; try { const answer = await callYarAPI( prompt, { loadingText: "در حال ساخت آگهی..." } ); if (result) { result.textContent = answer; } else { addChatMessage( answer, "ai" ); } } catch (error) { if (result) { result.textContent = "❌ " + ( error?.message || "ساخت آگهی انجام نشد." ); } else { showAIError(error); } } } // ===================================================== // 12. اتصال خودکار دکمه مترجم // ===================================================== function setupTranslator() { const button = document.getElementById( "translateButton" ); if (!button) { return; } button.addEventListener( "click", translateText ); } // ===================================================== // 13. اتصال خودکار کمک درسی // ===================================================== function setupStudy() { const button = document.getElementById( "studyButton" ); if (!button) { return; } button.addEventListener( "click", solveStudyQuestion ); } // ===================================================== // 14. اتصال خودکار نوشتن // ===================================================== function setupWriting() { const button = document.getElementById( "writingButton" ); if (!button) { return; } button.addEventListener( "click", generateWriting ); } // ===================================================== // 15. اتصال خودکار آگهی // ===================================================== function setupAdvertisement() { const button = document.getElementById( "adButton" ); if (!button) { return; } button.addEventListener( "click", generateAdvertisement ); } // ===================================================== // 16. اتصال فرم چت // ===================================================== function setupChat() { const form = document.getElementById( "chatForm" ); const input = document.getElementById( "messageInput" ); const sendButton = document.getElementById( "sendButton" ); if (!form || !input) { console.warn( "Chat form or message input not found." ); return; } form.addEventListener( "submit", async event => { event.preventDefault(); const message = input.value.trim(); if (!message) { return; } input.value = ""; if (sendButton) { sendButton.disabled = true; } try { await sendMessageToAI( message ); } finally { if (sendButton) { sendButton.disabled = false; } input.focus(); } } ); // Enter = ارسال // Shift + Enter = خط جدید input.addEventListener( "keydown", event => { if ( event.key === "Enter" && !event.shiftKey ) { event.preventDefault(); form.requestSubmit(); } } ); } // ===================================================== // 17. اجرای همه بخش‌ها // ===================================================== document.addEventListener( "DOMContentLoaded", () => { setupChat(); setupTranslator(); setupStudy(); setupWriting(); setupAdvertisement(); console.log( "Yar Afghanistan AI system initialized." ); } ); // ===================================================== // 18. در دسترس قرار دادن توابع برای HTML // ===================================================== // // اگر در HTML از onclick استفاده کرده باشی، // این توابع همچنان کار خواهند کرد. // // مثال: // onclick="translateText()" // onclick="solveStudyQuestion()" // onclick="generateWriting()" // onclick="generateAdvertisement()" // ===================================================== window.sendMessageToAI = sendMessageToAI; window.translateText = translateText; window.solveStudyQuestion = solveStudyQuestion; window.generateWriting = generateWriting; window.generateAdvertisement = generateAdvertisement; window.callYarAPI = callYarAPI; window.cleanAIResponse = cleanAIResponse; // ===================================================== // END // =====================================================
