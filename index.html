// ===================================================== // YAR AFGHANISTAN - AI CHAT // فقط پاسخ نهایی AI نمایش داده می‌شود // ===================================================== const CHAT_API_URL = "/api/chat"; // ----------------------------------------------------- // حذف Thinking / Reasoning از پاسخ AI // ----------------------------------------------------- function cleanAIResponse(text) { if (!text || typeof text !== "string") { return ""; } let result = text; // حذف بلوک‌های ... result = result.replace( //gi, "" ); // حذف بلوک‌های ... result = result.replace( //gi, "" ); // حذف بلوک‌های ... result = result.replace( //gi, "" ); // حذف بلوک‌های ... result = result.replace( //gi, "" ); // اگر پاسخ با "Thinking:" شروع شده باشد، // خطوط مربوط به thinking را حذف می‌کنیم. result = result.replace( /^\s*(thinking|reasoning|analysis)\s*:\s*[\s\S]*?(?=\n\s*(answer|final|پاسخ)\s*:)/i, "" ); // حذف برچسب‌های Answer / Final در ابتدای پاسخ result = result.replace( /^\s*(answer|final|پاسخ نهایی|پاسخ)\s*:\s*/i, "" ); // حذف کدهای Markdown اضافی در ابتدا و انتها result = result.replace( /^```(?:text|markdown)?\s*/i, "" ); result = result.replace( /\s*```$/i, "" ); return result.trim(); } // ----------------------------------------------------- // نمایش پیام در چت // ----------------------------------------------------- function addChatMessage(text, sender = "ai") { const chatContainer = document.getElementById("chatMessages") || document.getElementById("messages") || document.querySelector(".chat-messages"); if (!chatContainer) { console.error( "Chat container not found." ); return null; } const messageElement = document.createElement("div"); messageElement.className = sender === "user" ? "message user-message" : "message ai-message"; const content = document.createElement("div"); content.className = "message-content"; // برای امنیت، HTML خام AI را مستقیماً وارد DOM نمی‌کنیم. content.textContent = text; messageElement.appendChild(content); chatContainer.appendChild( messageElement ); // اسکرول به آخر چت chatContainer.scrollTop = chatContainer.scrollHeight; return messageElement; } // ----------------------------------------------------- // پیام "در حال پاسخ" // ----------------------------------------------------- function addLoadingMessage() { const chatContainer = document.getElementById("chatMessages") || document.getElementById("messages") || document.querySelector(".chat-messages"); if (!chatContainer) { return null; } const loading = document.createElement("div"); loading.className = "message ai-message loading-message"; loading.innerHTML = `
در حال پاسخ...
`; chatContainer.appendChild( loading ); chatContainer.scrollTop = chatContainer.scrollHeight; return loading; } // ----------------------------------------------------- // ارسال پیام به API // ----------------------------------------------------- async function sendMessageToAI(message) { const userMessage = typeof message === "string" ? message.trim() : ""; if (!userMessage) { return; } // نمایش پیام کاربر addChatMessage( userMessage, "user" ); // نمایش loading const loading = addLoadingMessage(); try { // ----------------------------------------------- // درخواست مستقیم به Cloudflare Pages Function // ----------------------------------------------- const response = await fetch( CHAT_API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: userMessage }) } ); // ----------------------------------------------- // دریافت JSON // ----------------------------------------------- let data; try { data = await response.json(); } catch { throw new Error( "پاسخ API معتبر نیست." ); } // حذف loading if (loading) { loading.remove(); } // ----------------------------------------------- // خطای API // ----------------------------------------------- if ( !response.ok || data.success === false ) { const errorMessage = data?.error || "ارتباط با هوش مصنوعی برقرار نشد."; addChatMessage( "❌ " + errorMessage, "ai" ); console.error( "Yar API error:", data ); return; } // ----------------------------------------------- // فقط reply یا message // ----------------------------------------------- let aiResponse = data?.reply; if ( typeof aiResponse !== "string" || !aiResponse.trim() ) { aiResponse = data?.message; } if ( typeof aiResponse !== "string" ) { addChatMessage( "❌ هوش مصنوعی پاسخ قابل استفاده‌ای برنگرداند.", "ai" ); return; } // ----------------------------------------------- // پاک‌سازی Thinking // ----------------------------------------------- aiResponse = cleanAIResponse( aiResponse ); // ----------------------------------------------- // اگر بعد از پاک‌سازی خالی شد // ----------------------------------------------- if (!aiResponse) { addChatMessage( "متأسفم، پاسخی دریافت نشد. لطفاً دوباره تلاش کنید.", "ai" ); return; } // ----------------------------------------------- // نمایش فقط پاسخ نهایی // ----------------------------------------------- addChatMessage( aiResponse, "ai" ); } catch (error) { // حذف loading if (loading) { loading.remove(); } console.error( "Yar AI connection error:", error ); addChatMessage( "❌ ارتباط با هوش مصنوعی برقرار نشد. لطفاً اینترنت و API را بررسی کنید.", "ai" ); } } // ===================================================== // اتصال فرم چت // ===================================================== document.addEventListener( "DOMContentLoaded", () => { const form = document.getElementById( "chatForm" ); const input = document.getElementById( "messageInput" ); const sendButton = document.getElementById( "sendButton" ); if (!form || !input) { console.warn( "Chat form or message input not found." ); return; } // ----------------------------------------------- // ارسال با فرم // ----------------------------------------------- form.addEventListener( "submit", async (event) => { event.preventDefault(); const message = input.value.trim(); if (!message) { return; } input.value = ""; if (sendButton) { sendButton.disabled = true; } try { await sendMessageToAI( message ); } finally { if (sendButton) { sendButton.disabled = false; } input.focus(); } } ); // ----------------------------------------------- // ارسال با Enter // ----------------------------------------------- input.addEventListener( "keydown", (event) => { if ( event.key === "Enter" && !event.shiftKey ) { event.preventDefault(); form.requestSubmit(); } } ); } );
